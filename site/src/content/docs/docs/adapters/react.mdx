```bash
pnpm add @gait/core @gait/react @gait/browser
```

The adapter uses `useSyncExternalStore`. It creates the Core instance once, publishes only snapshots and destroys it on unmount.

<Tabs syncKey="framework">
  <TabItem label="Browser">

Use a client-only factory to build a `GaitConfig` with `createBrowserEventSource()` and `createBrowserClock()`.

  </TabItem>
  <TabItem label="React">

```tsx
import type { GaitConfig } from '@gait/core';
import { useGait, useGaitSnapshot } from '@gait/react';

export function Metrics({ config }: { config: GaitConfig }) {
  const gait = useGait({ ...config, autoStart: true });
  const snapshot = useGaitSnapshot(gait);
  return <output>{snapshot.signals.keyboard.dwellTimeMs?.mean ?? '—'}</output>;
}
```

  </TabItem>

Keep the config creation outside server rendering. `autoStart: false` lets a session controller own the start boundary.
