Install the Core and the adapter for the environment you are integrating:

```bash
pnpm add @gait/core @gait/browser
```

The browser factory installs no listeners until `start()` is called. Snapshots are published when you explicitly call `flush()`.

```ts
import { createBrowserGait } from '@gait/browser';

const gait = createBrowserGait({
  signals: { keyboard: true, pointer: true, touch: true, wheel: true, context: true },
});
const unsubscribe = gait.subscribe((snapshot) => {
  console.log(snapshot.signals.keyboard.dwellTimeMs?.mean);
});

gait.start();
const timer = window.setInterval(() => gait.flush(), 250);

// At the end of the session or component lifetime:
window.clearInterval(timer);
unsubscribe();
gait.destroy();
```

  <TabItem label="React">

```tsx
import { useGait, useGaitSnapshot } from '@gait/react';

export function Metrics({ config }: { config: GaitConfig }) {
  const gait = useGait({ ...config, autoStart: true });
  const snapshot = useGaitSnapshot(gait);
  return <output>{snapshot.signals.pointer.speedPxPerMs?.mean ?? '—'}</output>;
}
```
